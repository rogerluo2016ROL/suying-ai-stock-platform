## SQL：概念补筛模板

```sql
-- 对所有 8+6 产业统一使用：行业直筛 + 概念补筛，UNION 去重
-- 例：集成电路方向

WITH industry_stocks AS (
    SELECT code, name, industry
    FROM stocks WHERE is_st = 0
      AND industry IN ('半导体','元器件','IT设备')
),
concept_stocks AS (
    SELECT DISTINCT m.con_code AS code, s.name, s.industry
    FROM ths_member m
    JOIN stocks s ON s.code = m.con_code AND s.is_st = 0
    WHERE m.ts_code IN ('885756.TI','885897.TI','885864.TI','885925.TI',
                        '885908.TI','886009.TI','886042.TI','885959.TI')
)
SELECT * FROM industry_stocks
UNION SELECT * FROM concept_stocks;
```
