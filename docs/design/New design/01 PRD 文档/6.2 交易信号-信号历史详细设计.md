## 1. 页面定位

信号历史是交易信号菜单的**第二 Tab**（Tab key: `history`）。操盘手在此页面验证：

1. 当前评分在历史中的相对位置（极值/正常）
2. 评分趋势方向（上升→信号增强，下降→信号衰减）
3. 该级别信号的历史准确率（评估可靠性）

## 3. 核心数据结构（复用）

```typescript
// 见 6.0 交易信号详细设计.md §2.5

interface SignalHistoryPoint {
  date: string              // "2026-06-27"
  compositeScore: number    // 综合评分 0-100
  rating: SignalRating      // STRONG_BUY | BUY | HOLD | SELL | STRONG_SELL
  dimBreakdown: Record<DimName, number>
}

interface SignalHistory {
  tsCode: string
  points: SignalHistoryPoint[]
  hitRates: {
    STRONG_BUY: { count: number; accuracy: number; avgReturn: number }
    BUY:        { count: number; accuracy: number; avgReturn: number }
  }
}
```

## 5. ECharts 历史趋势图配置

```typescript
// 文件: components/signals/HistoryTrendChart.tsx

import type { EChartsOption } from 'echarts'

function buildHistoryOption(
  points: SignalHistoryPoint[],
  showDataZoom: boolean = true
): EChartsOption {
  
  const dates = points.map(p => {
    const [m, d] = p.date.slice(5).split('-')
    return `${parseInt(m)}/${parseInt(d)}`
  })
  const scores = points.map(p => p.compositeScore)
  const ratings = points.map(p => p.rating)
  
  // 根据评级给每个数据点着色
  const pointColors = ratings.map(r => {
    switch (r) {
      case 'STRONG_BUY': return '#ff4d4f'
      case 'BUY': return '#f5a623'
      case 'HOLD': return '#3d8bff'
      case 'SELL': return '#2ec27e'
      case 'STRONG_SELL': return '#5e6a7d'
    }
  })
  
  const option: EChartsOption = {
    backgroundColor: 'transparent',
    tooltip: {
      trigger: 'axis',
      backgroundColor: '#1b2330',
      borderColor: '#2a3444',
      textStyle: { fontSize: 11, color: '#e8edf4' },
      formatter: (params: any) => {
        const p = params[0]
        const idx = p.dataIndex
        const r = ratings[idx]
        const rl = r === 'STRONG_BUY' ? '强烈买入' : r === 'BUY' ? '买入' : r === 'HOLD' ? '持有' : r === 'SELL' ? '卖出' : '强烈卖出'
        return `${dates[idx]}<br/>评分: <b>${p.value}</b> 分 · 评级: <b>${rl}</b>`
      },
    },
    grid: { left: '4%', right: '4%', top: '8%', bottom: showDataZoom ? '18%' : '6%' },
    xAxis: {
      type: 'category',
      data: dates,
      axisLabel: { fontSize: 9, color: '#5e6a7d', rotate: 30 },
      axisLine: { lineStyle: { color: '#1f2733' } },
      axisTick: { show: false },
    },
    yAxis: {
      type: 'value',
      min: 0,
      max: 100,
      splitLine: { lineStyle: { color: '#1f2733' } },
      axisLabel: { fontSize: 10, color: '#5e6a7d' },
    },
    series: [{
      type: 'line',
      data: scores,
      name: '综合评分',
      smooth: true,
      lineStyle: { color: '#3d8bff', width: 2 },
      areaStyle: {
        color: {
          type: 'linear',
          x: 0, y: 0, x2: 0, y2: 1,
          colorStops: [
            { offset: 0, color: 'rgba(61,139,255,0.18)' },
            { offset: 1, color: 'rgba(61,139,255,0)' },
          ],
        },
      },
      symbol: 'circle',
      symbolSize: 5,
      itemStyle: { 
        color: (params: any) => pointColors[params.dataIndex],
      },
      markLine: {
        silent: true,
        symbol: 'none',
        data: [
          {
            yAxis: 80,
            lineStyle: { color: 'rgba(255,77,79,0.25)', type: 'dashed' },
            label: { formatter: '强烈买入 80', fontSize: 9, color: '#ff4d4f' },
          },
          {
            yAxis: 60,
            lineStyle: { color: 'rgba(245,166,35,0.25)', type: 'dashed' },
            label: { formatter: '买入 60', fontSize: 9, color: '#f5a623' },
          },
          {
            yAxis: 40,
            lineStyle: { color: 'rgba(94,106,125,0.25)', type: 'dashed' },
            label: { formatter: '卖出 40', fontSize: 9, color: '#5e6a7d' },
          },
        ],
      },
    }],
    dataZoom: showDataZoom ? [
      { type: 'inside' },
      {
        type: 'slider',
        height: 18,
        bottom: 2,
        borderColor: '#1f2733',
        backgroundColor: '#11161f',
        fillerColor: 'rgba(61,139,255,0.1)',
        textStyle: { fontSize: 9, color: '#5e6a7d' },
      },
    ] : [],
  }
  
  return option
}
```

## 6. 命中率面板（HitRatePanel）

```typescript
// 文件: components/signals/HitRatePanel.tsx

interface HitRatePanelProps {
  hitRates: SignalHistory['hitRates']
  recentCount?: { strongBuy: number; fulfilled: number } // 近30天兑现统计
}

function HitRatePanel({ hitRates, recentCount }: HitRatePanelProps) {
  return (
    <div>
      <div className="hit">
        <HitRateCard
          level="STRONG_BUY"
          icon="🟢"
          color="var(--up)"
          data={hitRates.STRONG_BUY}
        />
        <HitRateCard
          level="BUY"
          icon="🟡"
          color="var(--warn)"
          data={hitRates.BUY}
        />
      </div>
      
      {recentCount && (
        <div className="hit-summary">
          近30天 STRONG_BUY 信号出现 <b>{recentCount.strongBuy}</b> 次,
          其中 <b style={{ color: 'var(--up)' }}>{recentCount.fulfilled}</b> 次兑现
        </div>
      )}
    </div>
  )
}

function HitRateCard({ level, icon, color, data }: {
  level: string; icon: string; color: string;
  data: { count: number; accuracy: number; avgReturn: number }
}) {
  const retColor = data.avgReturn >= 0 ? 'var(--up)' : 'var(--down)'
  const retSign = data.avgReturn >= 0 ? '+' : ''
  
  return (
    <div className="hc">
      <div className="hl" style={{ color }}>{icon} {level}</div>
      <div className="hm">
        出现 <b>{data.count}</b> 次<br/>
        方向正确率 <b style={{ color }}>{data.accuracy}%</b><br/>
        平均收益 <b style={{ color: retColor }}>{retSign}{data.avgReturn}%</b>
      </div>
    </div>
  )
}
```

## 7. 数据流

```
SignalDetailPage (父级)
  │
  └── activeTab === 'history' → <SignalHistoryTab
        signalHistory={signalHistory}
        userRole={userRole}
      />

数据来源：
  GET /api/v1/signal/history?code={tsCode}&days=30
  
响应 (SignalHistory):
  {
    tsCode: "300750.SZ",
    points: [
      { date: "2026-05-28", compositeScore: 52, rating: "HOLD", dimBreakdown: {...} },
      ...
      { date: "2026-06-27", compositeScore: 85, rating: "STRONG_BUY", dimBreakdown: {...} }
    ],
    hitRates: {
      STRONG_BUY: { count: 15, accuracy: 80, avgReturn: 5.2 },
      BUY: { count: 8, accuracy: 68, avgReturn: 2.1 }
    }
  }

加载态：
  - 图表区域：灰色骨架屏 + Spin 居中
  - 命中率面板：2 个灰色卡片占位

错误态：
  - 图表加载失败 → Card 内 Alert "历史数据加载失败" + 重试
  - 命中率数据缺失 → "暂无历史命中率数据"

空态：
  - 数据不足 3 天 → 提示 "历史数据不足，需至少3个交易日"
```

## 平台化优化补充

> 继承自 `0.3 平台化云端多租户总体需求设计.md`、`0.4 平台化云端多券商架构设计.md`、`0.5 平台化详细方案设计.md`。

| 维度 | 设计要求 |
|---|---|
| 模块定位 | 交易信号：公共信号规则与用户私有订阅、处理状态分离。 |
| 角色差异 | 操盘手保留专业工作台；个人投资者默认简化解释视图；系统管理员获得配置、审计和运行治理视图。 |
| 数据隔离 | 公共信号 + 私有已读/订阅/处置。所有私有对象必须写入 `tenantId / ownerUserId / accountId? / visibility / dataScope`。 |
| 云端部署 | API 必须从认证上下文恢复租户与用户身份，禁止前端传入越权租户；列表接口默认只返回当前身份可见数据。 |
| 券商/交易 | 信号只作为订单依据之一。凡涉及资金、订单、策略启停的动作必须经过 `RiskGate` 并写入 append-only 审计。 |
| 原型要求 | 页面顶部展示角色视图、数据域、账户隔离、券商模式和云端就绪状态；不得把公共行情数据与私有交易数据混在同一区块而不标识。 |
